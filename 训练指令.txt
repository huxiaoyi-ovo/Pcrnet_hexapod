# Mix Gate 微调（S3 Doorway / S4 Crossing / S5 Transition 混合）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill moe --task hex_mix_gate --low_level_ckpt <LOW_LEVEL_CKPT> --follow_ckpt <FOLLOW_CKPT> --avoid_ckpt <AVOID_CKPT> --finetune_from <GATE_S4_CKPT> --num_envs 512 --num_steps 48 --num_iterations 120 --decimation 5 --headless

# Gate + T1 弱遮挡（低概率，训练稳定后再启）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill moe --task hex_mix_gate --low_level_ckpt <LOW_LEVEL_CKPT> --follow_ckpt <FOLLOW_CKPT> --avoid_ckpt <AVOID_CKPT> --finetune_from <GATE_S4_CKPT> --t1_goal_occlusion --t1_goal_occlusion_prob 0.02 --t1_goal_occlusion_len 10 --num_envs 512 --num_steps 48 --num_iterations 120 --decimation 5 --headless

# Resume / Finetune 示例

python legged_gym/scripts/train_highlevel.py --mode teacher --skill avoid --task hex_s1 --low_level_ckpt <LOW_LEVEL_CKPT> --finetune_from <AVOID_CKPT> --num_envs 512 --num_steps 24 --num_iterations 120 --decimation 5 --headless
