# ------------------------
# 高层 V5 Teacher 训练（cmd-space MoE）
# ------------------------
# 训练场景说明（S1–S6）
# - S1 Corridor：走廊 + 门洞收缩段，通道内少量离散障碍，窄口能力训练
# - S2 Forest：开放平地随机静态障碍（pole/block 混合），Follow/Avoid 主训练分布
# - S3 Doorway：房间-门洞拓扑，门洞宽度/位置随机，门口+转向冲突
# - S4 Crossing：动态横穿障碍，速度由反应窗口步数反推（冲突最强）
# - S5 Transition：稀疏→密集（或反向）分段密度切换，Gate 情境切换
# - S6 OOD：结构化 OOD（cluster / nonconvex / maze）混合采样；如需固定模板，修改 hex_s6_config.py 的 ood_template

# S2 Follow vs Avoid 说明：
# - S2 场景相同，差异来自 skill 与奖励/门控（Follow 更偏跟随，Avoid 更偏风险约束）。

# S2 Follow（Forest，Follow 主训练分布）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill follow --task hex_s2 --low_level_ckpt <LOW_LEVEL_CKPT> --num_envs 512 --num_steps 24 --num_iterations 200 --decimation 5 --headless

# S2 Avoid（Forest 主训）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill avoid --task hex_s2 --low_level_ckpt <LOW_LEVEL_CKPT> --num_envs 512 --num_steps 24 --num_iterations 200 --decimation 5 --headless

# S1 Avoid 微调（Corridor，跨 task finetune）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill avoid --task hex_s1 --low_level_ckpt <LOW_LEVEL_CKPT> --finetune_from <AVOID_FOREST_CKPT> --num_envs 512 --num_steps 24 --num_iterations 120 --decimation 5 --headless

# S4 Gate 单场景训练（Crossing，冻结 Follow/Avoid）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill moe --task hex_s4 --low_level_ckpt <LOW_LEVEL_CKPT> --follow_ckpt <FOLLOW_CKPT> --avoid_ckpt <AVOID_CKPT> --num_envs 512 --num_steps 48 --num_iterations 200 --decimation 5 --headless

# Mix Gate 微调（S3 Doorway / S4 Crossing / S5 Transition 混合）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill moe --task hex_mix_gate --low_level_ckpt <LOW_LEVEL_CKPT> --follow_ckpt <FOLLOW_CKPT> --avoid_ckpt <AVOID_CKPT> --finetune_from <GATE_S4_CKPT> --num_envs 512 --num_steps 48 --num_iterations 120 --decimation 5 --headless

# Gate + T1 弱遮挡（低概率，训练稳定后再启）
python legged_gym/scripts/train_highlevel.py --mode teacher --skill moe --task hex_mix_gate --low_level_ckpt <LOW_LEVEL_CKPT> --follow_ckpt <FOLLOW_CKPT> --avoid_ckpt <AVOID_CKPT> --finetune_from <GATE_S4_CKPT> --t1_goal_occlusion --t1_goal_occlusion_prob 0.02 --t1_goal_occlusion_len 10 --num_envs 512 --num_steps 48 --num_iterations 120 --decimation 5 --headless
